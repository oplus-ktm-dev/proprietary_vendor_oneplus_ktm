# NCI CORE RESET CMD
send 20000100

# NCI CORE INIT CMD
send 2001020000


# NCI SYSTEM ENABLE PROPRIETARY CMD
send 2F0200

# NCI Disable standby mode
send 2F000100

# Start Continuous RF ON
send 2F3D022001

# 查TXLDO电流
exec 2F320108 4F32
